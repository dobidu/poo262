// Classe QObject: macros Q_OBJECT habilita signals/slots e reflection
class mixer_window : public QMainWindow {
    Q_OBJECT  // macro obrigatória para signals/slots
public:
    explicit mixer_window(audio_engine& engine, QWidget* parent = nullptr);

signals:  // declarações — implementação gerada pelo MOC
    void play_requested();
    void volume_changed(float volume);

private slots:  // recebe signals
    void on_play_clicked()          { emit play_requested(); }
    void on_volume_slider(int val)  { emit volume_changed(val / 100.0f); }

private:
    audio_engine& engine_;  // referência — não possui
};

// Conectar signals a slots: sem acoplamento direto entre objetos
connect(&window,  &mixer_window::play_requested,
        &engine,  &audio_engine::play);
connect(&window,  &mixer_window::volume_changed,
        &engine,  &audio_engine::set_master_volume);
// Quando window emite play_requested → engine.play() é chamado
// window não sabe que engine existe; engine não sabe sobre window

int main(int argc, char* argv[]) {
    QApplication app{argc, argv};

    // Domínio: sem dependência de Qt
    json_repository repo;
    audio_engine engine{repo};  // DIP: engine depende da abstração

    // Apresentação: recebe engine por referência
    mixer_window window{engine};
    window.show();

    return app.exec();
}

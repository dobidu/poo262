Factory Method e Singleton

cpp Factory Method — effect_factory 
// Factory Method: cria objetos sem expor o tipo concreto
class effect_factory {
public:
    // Cria pelo nome — extensível sem modificar o código chamador
    static std::unique_ptr create(const std::string& type) {
        if (type == "gain")   return std::make_unique(1.0f);
        if (type == "reverb") return std::make_unique(0.5f, 0.3f);
        if (type == "delay")  return std::make_unique(250.0f, 0.5f, 0.3f);
        throw std::invalid_argument{"efeito desconhecido: " + type};
    }
    // Overloads convenience com parâmetros
    static auto make_gain  (float f)           { return std::make_unique(f); }
    static auto make_reverb(float r, float w)  { return std::make_unique(r,w); }
};

// Singleton — audio_config (configuração global única)
class audio_config {
public:
    static audio_config& instance() {
        static audio_config inst;  // C++11: thread-safe, construído uma vez
        return inst;
    }
    float master_volume() const noexcept { return vol_; }
    void set_master_volume(float v) {
        if (v < 0.0f || v > 2.0f) throw std::domain_error{"volume fora do range"};
        vol_ = v;
    }
private:
    audio_config() = default;  // privado: impede instanciação direta
    float vol_ = 1.0f;
};

Decorator e Composite

cpp Decorator e Composite na cadeia de efeitos 
// DECORATOR: adiciona comportamento sem modificar a classe base
class metering_decorator : public effect {
    std::unique_ptr inner_;
    float last_in_peak_  = 0.0f;
    float last_out_peak_ = 0.0f;
public:
    explicit metering_decorator(std::unique_ptr e)
        : effect{e->name() + "_metered"}, inner_{std::move(e)} {}

    void process(audio_buffer& buf) override {
        last_in_peak_ = buf.peak_level();
        inner_->process(buf);           // delega para o efeito real
        last_out_peak_ = buf.peak_level();
    }
    float last_input_peak()  const noexcept { return last_in_peak_; }
    float last_output_peak() const noexcept { return last_out_peak_; }
    std::unique_ptr clone() const override {
        return std::make_unique(inner_->clone());
    }
};

// COMPOSITE: trata grupo de efeitos como um único efeito
class effect_group : public effect {
    std::vector> effects_;
public:
    explicit effect_group(std::string name) : effect{std::move(name)} {}
    void add(std::unique_ptr e) { effects_.push_back(std::move(e)); }
    void process(audio_buffer& buf) override {
        for (auto& e : effects_) if (e->enabled()) e->process(buf);
    }
    std::unique_ptr clone() const override; // deep clone
};

Observer e Strategy

cpp Observer — transport notifica UI 
// Observer: transport notifica listeners sem acoplamento direto
class transport_observer {
public:
    virtual void on_play()           = 0;
    virtual void on_stop()           = 0;
    virtual void on_seek(double pos) = 0;
    virtual ~transport_observer()    = default;
};

class transport {
    std::vector observers_;
public:
    void add_observer(transport_observer* o)    { observers_.push_back(o); }
    void remove_observer(transport_observer* o) {
        observers_.erase(std::remove(observers_.begin(), observers_.end(), o),
                         observers_.end());
    }
    void play() {
        // Notifica todos os observers — transport não sabe quem são
        for (auto* o : observers_) o->on_play();
    }
};

// Qualquer objeto pode ser observer (UI, logger, gravador...)
class ui_controller : public transport_observer {
    void on_play()           override { update_play_button(); }
    void on_stop()           override { update_stop_button(); }
    void on_seek(double pos) override { update_position_display(pos); }
};

Prompt sugerido

"Implemente o padrão Chain of Responsibility para a cadeia de efeitos: cada efeito decide se processa o buffer ou passa para o próximo. Como isso difere do Composite implementado no Sintonia?"

⚠️
O que o LLM costuma errar

LLMs frequentemente confundem padrões: Decorator com Proxy, Composite com Facade, Factory Method com Abstract Factory. Peça que o LLM explique a diferença entre dois padrões antes de implementar.
